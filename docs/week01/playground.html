class CircleRotate(Scene):
    def construct(self):
        center = LEFT * 3.2 + DOWN * 0.1
        theta = ValueTracker(0)                 # 회전 각도
        people = VGroup()
        for i, l in enumerate("ABCD"):          # ← "ABCDE"로 바꾸면 5명
            p = person(l, RED if l == "A" else BLUE)
            p.add_updater(lambda m, i=i: m.move_to(
                seat(center, 2.0, i, 4, theta.get_value())))
            people.add(p)
        self.play(FadeIn(table(center)), LaggedStart(*[GrowFromCenter(p) for p in people]))
        for k in range(1, 4):
            self.play(theta.animate.set_value(k * PI / 2))   # 90°씩 회전

로컬 실행: python -m manim -pqm scripts/w01_anim.py CircleRotate · 전체 코드 scripts/w01_anim.py
